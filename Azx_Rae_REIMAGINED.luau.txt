local function getCharacter()
    return LocalPlayer.Character
end

local function getHRP()
    local char = getCharacter()
    return char and char:FindFirstChild('HumanoidRootPart')
end

local function getHumanoid()
    local char = getCharacter()
    return char and char:FindFirstChildOfClass('Humanoid')
end

local function getTool()
    local char = getCharacter()
    return char and char:FindFirstChildOfClass('Tool')
end

local function getToolHandle()
    local tool = getTool()
    return tool and (tool:FindFirstChild('Handle') or tool:FindFirstChildWhichIsA('BasePart'))
end

local function getClosestPlayer(maxDist)
    maxDist = maxDist or Cfg.maxRange
    local hrp    = getHRP()
    if not hrp then return nil end
    local best, bestDist = nil, maxDist
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and plr.Character then
            local targetHRP = plr.Character:FindFirstChild('HumanoidRootPart')
            if targetHRP then
                local dist = (hrp.Position - targetHRP.Position).Magnitude
                if dist < bestDist then
                    local hum = plr.Character:FindFirstChildOfClass('Humanoid')
                    if hum and hum.Health > 0 then
                        best     = plr
                        bestDist = dist
                    end
                end
            end
        end
    end
    return best
end

local function fireTouchOnCharacter(handle, targetChar)
    for _, part in ipairs(targetChar:GetDescendants()) do
        if part:IsA('BasePart') and part.Parent then
            firetouchinterest(handle, part, 0)
            firetouchinterest(handle, part, 1)
        end
    end
end

-- AIMBOT (Faces nearest enemy, moves highlight)
-- ══════════════════════════════════════════════════════════════════════
local function runAimbot()
    local char = getCharacter()
    local hrp  = getHRP()
    if not (char and hrp and Cfg.AimbotEnabled) then return end

    local myPos = hrp.Position
    local candidates = {}

    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and plr.Character then
            local targetHRP = plr.Character:FindFirstChild('HumanoidRootPart')
            if targetHRP then
                local dist = (targetHRP.Position - myPos).Magnitude
                if dist <= Cfg.maxRange then
                    local hum = plr.Character:FindFirstChildOfClass('Humanoid')
                    if hum and hum.Health > 0 then
                        table.insert(candidates, { Part = targetHRP, Distance = dist })
                    end
                end
            end
        end
    end

    local closestPart, closestDist = nil, math.huge
    for _, entry in ipairs(candidates) do
        if entry.Distance < closestDist then
            closestPart = entry.Part
            closestDist = entry.Distance
        end
    end

    if closestPart then
        HighlightPart.CFrame = CFrame.new(closestPart.Position)

        local dir    = (closestPart.Position - myPos).Unit
        local flatDir = Vector3.new(dir.X, 0, dir.Z).Unit

        -- Wiggle mode
        local wx, wz = 0, 0
        if Cfg.Wiggle then
            if Cfg.WiggleMode == 'Right' then
                wz = math.random(0, 1)
            elseif Cfg.WiggleMode == 'Left' then
                wx = math.random(0, 1)
            elseif Cfg.WiggleMode == 'Both' then
                wx, wz = math.random(0, 1), math.random(0, 1)
            end
            flatDir = Vector3.new(flatDir.X + wx, 0, flatDir.Z + wz).Unit
        end

        hrp.CFrame = CFrame.new(myPos) * CFrame.lookAt(Vector3.new(), flatDir)

        -- Auto Swing
        local tool = getTool()
        if tool and Cfg.AutoSwing then
            tool:Activate()
        end

        -- Teleport strafe (hold T)
        if UserInputService:IsKeyDown(Enum.KeyCode.T) then
            local tpDist = Cfg.TPDistance
            if Cfg.randomizeTeleport then
                tpDist = tpDist + math.random(-2, 2)
            end
            local tpPos   = closestPart.Position - closestPart.CFrame.LookVector * tpDist
            local tpDir   = (closestPart.Position - tpPos).Unit
            local flatTp  = Vector3.new(tpDir.X, 0, tpDir.Z).Unit
            hrp.CFrame    = CFrame.new(tpPos) * CFrame.lookAt(Vector3.new(), flatTp)
        end
    else
        HighlightPart.CFrame = CFrame.new(0, -1000, 0)
    end
end

RunService.Heartbeat:Connect(function()
    runAimbot()
end)

local killAuraThrottle = 0

    local char   = getCharacter()
    local hrp    = getHRP()
    local handle = getToolHandle()
    if not (char and hrp and handle) then return end

    local now = tick()

    killAuraThrottle = now

    local myPos = hrp.Position
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and plr.Character then
            local targetHRP = plr.Character:FindFirstChild('HumanoidRootPart')
            if targetHRP then
                local dist = (targetHRP.Position - myPos).Magnitude
                if dist <= Cfg.maxRange then
                    local hum = plr.Character:FindFirstChildOfClass('Humanoid')
                    if hum and hum.Health > 0 then
                        task.spawn(function()
                            fireTouchOnCharacter(handle, plr.Character)
                        end)
                        local tool = getTool()
                        if tool then tool:Activate() end
                    end
                end
            end
        end
    end
end)

-- ══════════════════════════════════════════════════════════════════════
-- BODY LOCK & PREDICTION
-- ══════════════════════════════════════════════════════════════════════
RunService.RenderStepped:Connect(function()
    local char = getCharacter()
    local hrp  = getHRP()
    local hum  = getHumanoid()
    if not (char and hrp and hum) then return end

    local target     = getClosestPlayer()
    local targetChar = target and target.Character
    local targetPart = targetChar and targetChar:FindFirstChild(Cfg.LockBodyPart)

    if Cfg.BodyLock and targetPart then
        local aimPos = targetPart.Position
        if Cfg.Prediction then
            local targetHRP = targetChar:FindFirstChild('HumanoidRootPart')
            if targetHRP then
                aimPos = aimPos + targetHRP.AssemblyLinearVelocity * Cfg.PredictionOffset
            end
        end
        hrp.CFrame      = CFrame.new(hrp.Position, Vector3.new(aimPos.X, hrp.Position.Y, aimPos.Z))
        hum.AutoRotate  = false
    else
        hum.AutoRotate  = true
    end

    -- Auto Hit
    if Cfg.AutoHit and targetPart and (hrp.Position - targetPart.Position).Magnitude <= Cfg.AutoHitStuds then
        local tool = getTool()
        if tool and tool:FindFirstChild('Handle') then
            tool:Activate()
        end
    end

    -- Spin
    if Cfg.SpinEnabled then
        hrp.CFrame = hrp.CFrame * CFrame.Angles(0, math.rad(Cfg.SpinSpeed), 0)
    end

    -- Auto Jump near players
    if Cfg.AutoJump and targetPart and (hrp.Position - targetPart.Position).Magnitude <= Cfg.JumpStuds then
        if hum.FloorMaterial ~= Enum.Material.Air then
            hum:ChangeState(Enum.HumanoidStateType.Jumping)
        end
    end
end)

RunService.RenderStepped:Connect(function()
    local hrp = getHRP()
    if not hrp then
        for _, seg in ipairs(RingSegments) do
            seg.Transparency = 1
        end
        return
    end

    RingHue   = (RingHue   + 0.005) % 1
    RingAngle = (RingAngle + 1)

    local color   = Color3.fromHSV(RingHue, 1, 1)
    local showRing = Cfg.AimbotEnabled -- show ring when aimbot on

    for i, seg in ipairs(RingSegments) do
        local angle = (i / #RingSegments) * math.pi * 2 + math.rad(RingAngle)
        local rx    = math.cos(angle) * RingRadius
        local rz    = math.sin(angle) * RingRadius
        seg.Position    = hrp.Position + Vector3.new(rx, -2.8, rz)
        seg.Color       = color
        seg.Transparency = showRing and 0.2 or 1
    end
end)

-- SERVER HOP
-- ══════════════════════════════════════════════════════════════════════
local function serverHop(sizeType)
    local ok, data = pcall(function()
        return HttpService:JSONDecode(
            game:HttpGet('https://games.roblox.com/v1/games/' .. game.PlaceId .. '/servers/Public?sortOrder=Desc&limit=100')
        )
    end)
    if ok and data and data.data then
        for _, server in ipairs(data.data) do
            if server.playing < server.maxPlayers and server.id ~= game.JobId then
                local p = server.playing
                local match = false
                if sizeType == 'Small'  and (p >= 3  and p <= 5)  then match = true end
                if sizeType == 'Medium' and (p >= 13 and p <= 17) then match = true end
                if sizeType == 'Large'  and (p >= 20 and p <= 23) then match = true end
                if sizeType == 'Any'    and p >= 1                 then match = true end
                if match then
                    TeleportService:TeleportToPlaceInstance(game.PlaceId, server.id, LocalPlayer)
                    return
                end
            end
        end
    end
    warn('[AzxRae] No suitable server found for type: ' .. sizeType)
end

-- ══════════════════════════════════════════════════════════════════════
-- AUTO SNIPE (teleport + attack players with most time)
-- ══════════════════════════════════════════════════════════════════════
local snipeTarget = nil
local snipeDebounce = {}

local function isOnSpawn(plr)
    if not plr.Character then return false end
    local hrp = plr.Character:FindFirstChild('HumanoidRootPart')
    if not hrp then return false end
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Blacklist
    params.FilterDescendantsInstances = { plr.Character }
    local result = workspace:Raycast(hrp.Position, Vector3.new(0, -6, 0), params)
    if result then
        local spawnFolder = workspace:FindFirstChild('Terrain') and workspace.Terrain:FindFirstChild('Spawn')
        if spawnFolder then
            for _, part in ipairs(spawnFolder:GetDescendants()) do
                if part == result.Instance then return true end
            end
        end
    end
    return false
end

local function snipeTeleportAndAttack(targetPlr)
    local char      = getCharacter()
    local hrp       = getHRP()
    local targetChar = targetPlr.Character
    if not (char and hrp and targetChar) then return end
    local targetHRP = targetChar:FindFirstChild('HumanoidRootPart')
    if not targetHRP then return end
    hrp.CFrame = CFrame.new(targetHRP.Position) * CFrame.Angles(0, math.random(0, 360), 0)
    local tool  = LocalPlayer.Backpack:FindFirstChildOfClass('Tool')
    if tool then
        tool.Parent = char
        task.spawn(function()
            local start = tick()
            while tool and tool.Parent == char and Cfg.AutoSnipe do
                tool:Activate()
                task.wait()
                if tick() - start > 2 then break end
            end
        end)
    end
end

task.spawn(function()
    while task.wait(0) do
        if Cfg.AutoSnipe then
            local hrp = getHRP()
            if not hrp then task.wait(0.5) continue end
            local best, bestTime = nil, math.huge
            for _, plr in ipairs(Players:GetPlayers()) do
                if plr ~= LocalPlayer and plr.Character then
                    local ls   = plr:FindFirstChild('leaderstats')
                    local time = ls and ls:FindFirstChild('Time') and ls.Time.Value or 0
                    local pHRP = plr.Character:FindFirstChild('HumanoidRootPart')
                    local hum  = plr.Character:FindFirstChildOfClass('Humanoid')
                    if pHRP and hum and hum.Health > 0 and time >= Cfg.SnipeMinTime then
                        local dist = (hrp.Position - pHRP.Position).Magnitude
                        if (dist <= Cfg.SnipeMaxDist or Cfg.AutoSnipe) and not isOnSpawn(plr) then
                            if time < bestTime then
                                best     = plr
                                bestTime = time
                            end
                        end
                    end
                end
            end
            if best and best ~= snipeTarget then
                snipeTarget = best
                task.spawn(function()
                    while snipeTarget and Cfg.AutoSnipe do
                        local tChar = snipeTarget.Character
                        local tHum  = tChar and tChar:FindFirstChildOfClass('Humanoid')
                        if not (tChar and tHum and tHum.Health > 0) then
                            snipeTarget = nil
                            break
                        end
                        snipeTeleportAndAttack(snipeTarget)
                        task.wait(0)
                    end
                end)
            end
            if snipeTarget then
                local tChar = snipeTarget.Character
                local tHum  = tChar and tChar:FindFirstChildOfClass('Humanoid')
                if not (tChar and tHum and tHum.Health > 0) then
                    snipeTarget = nil
                end
            end
        end
        task.wait(0)
    end
end)

    if processed then return end
    local key = input.KeyCode

    if key == Enum.KeyCode.R then
        Cfg.AimbotEnabled = not Cfg.AimbotEnabled
        print('[AzxRae] Aimbot:', Cfg.AimbotEnabled)
    elseif key == Enum.KeyCode.P then
        Cfg.KillAuraEnabled = not Cfg.KillAuraEnabled
        print('[AzxRae] Kill Aura:', Cfg.KillAuraEnabled)
    elseif key == Enum.KeyCode.K then
        Cfg.AutoWalk = not Cfg.AutoWalk
        print('[AzxRae] Auto Walk:', Cfg.AutoWalk)
    elseif key == Enum.KeyCode.F then
        Cfg.AutoSwing = not Cfg.AutoSwing
        print('[AzxRae] Auto Swing:', Cfg.AutoSwing)
    elseif key == Enum.KeyCode.Q then
        Cfg.WalkBypassEnabled = not Cfg.WalkBypassEnabled
        print('[AzxRae] Walk Bypass:', Cfg.WalkBypassEnabled)
    elseif key == Enum.KeyCode.J then
        Cfg.InfJump = not Cfg.InfJump
        print('[AzxRae] Infinite Jump:', Cfg.InfJump)
    elseif key == Enum.KeyCode.G then
        Cfg.SpinEnabled = not Cfg.SpinEnabled
        print('[AzxRae] Spin:', Cfg.SpinEnabled)
    elseif key == Enum.KeyCode.H then
