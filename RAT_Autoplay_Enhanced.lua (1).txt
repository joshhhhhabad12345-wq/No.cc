local function pickKeyboardKey(bind)
    if type(bind) ~= "table" then return nil end
    local fallback
    for _, v in pairs(bind) do
        local name
        if type(v) == "string" then name = v
        elseif typeof(v) == "EnumItem" then name = v.Name end
        if name then
            local up = name:upper()
            if Enum.KeyCode[up] then
                if up ~= "LEFT" and up ~= "RIGHT" and up ~= "UP" and up ~= "DOWN" then
                    return up
                end
                fallback = fallback or up
            end
        end
    end
    return fallback
end

do
    local plr = Players.LocalPlayer
    local getPref = plr and plr:FindFirstChild("GetPref")
    if getPref then
        local ok, binds = pcallF(function() return getPref:Invoke("Binds") end)
        if ok and type(binds) == "table" then
            local laneNames = { "Left", "Down", "Up", "Right", "Extra1", "Extra2", "Extra3" }
            for i, n in ipairs(laneNames) do
                local k = pickKeyboardKey(binds[n])
                if k then settings.bindKey[i] = k end
            end
        end
    end
end

-- 7) RNG / Gauss (Box-Muller)
------------------------------------------------------------
local rng = Random.new(tick() * 1e3)

local function gauss()
    local u1 = mathMax(rng:NextNumber(), 1e-7)
    return mathSqrt(-2 * mathLog(u1)) * mathCos(2 * mathPi * rng:NextNumber())
end

local function jitterMs(sigma, clampMag)
    if sigma <= 0 then return 0 end
    local v = gauss() * sigma
    if v < -clampMag then return -clampMag end
    if v >  clampMag then return  clampMag end
    return v
end

-- FEATURE 7: Randomize humanizer within safe ranges
local SAFE_JITTER_MIN, SAFE_JITTER_MAX = 4, 9
local SAFE_BAD_MIN,    SAFE_BAD_MAX    = 0.5, 1.2

local function randomizeHumanizer()
    local j = SAFE_JITTER_MIN + rng:NextNumber() * (SAFE_JITTER_MAX - SAFE_JITTER_MIN)
    settings.pressJitterMs        = j
    settings.pressJitterClampMs   = j + 1
    settings.releaseJitterMs      = j
    settings.releaseJitterClampMs = j + 1
    settings.badChance = SAFE_BAD_MIN + rng:NextNumber() * (SAFE_BAD_MAX - SAFE_BAD_MIN)
end

-- FEATURE 5: snapshot current stats for history
local function snapshotSong()
    local entry = {
        name   = stats.songName,
        acc    = string.format("%.2f%%", (function()
            local t = stats.notesHit + stats.notesMissed + stats.notesBad
            if t == 0 then return 100.0 end
            return (stats.notesHit / t) * 100
        end)()),
        hits   = stats.notesHit,
        misses = stats.notesMissed,
        bads   = stats.notesBad,
        best   = stats.bestStreak,
    }
    table.insert(stats.songHistory, 1, entry)
    if #stats.songHistory > 5 then
        table.remove(stats.songHistory)
    end
end

local function resetStats(newSong)
    if newSong and stats.notesHit + stats.notesMissed + stats.notesBad > 5 then
        snapshotSong()
    end
    stats.notesHit       = 0
    stats.notesMissed    = 0
    stats.notesBad       = 0
    stats.minesSkipped   = 0
    stats.mustHitsHit    = 0
    stats.chords         = 0
    stats.streak         = 0
    stats.bestStreak     = 0
    stats.songCount      = stats.songCount + 1
    stats.recentHitTimes = {}
    stats.timingDeltas   = {}
    stats.avgDeltaMs     = 0
    stats.missPositions  = {}
    stats.lastTickLanes  = {}
    -- FEATURE 7/15: randomize on new song
    if settings.humanizerRandomize then randomizeHumanizer() end
    if settings.randomPresetPerSong then
        applyPreset(PRESETS[math.random(1, #PRESETS)])
    end
end

local function getAccuracy()
    local total = stats.notesHit + stats.notesMissed + stats.notesBad
    if total == 0 then return 100.0 end
    return (stats.notesHit / total) * 100
end

    vibratoTimers   = {}

local detectedLaneCount = 4

local function updateLaneCount(strums)
    local n = #strums
    if n > detectedLaneCount then detectedLaneCount = n end
end

local lastNoteCount = 0
local hadPlayer     = false
local lastSideIdx   = nil
local wasPaused     = false  -- FEATURE 18

local function releaseAllPressed(maxLanes)
    maxLanes = maxLanes or detectedLaneCount
    for laneIdx = 1, maxLanes do
        if heldSustains[laneIdx] then
            if lastSideIdx then
                pcallF(Trigger, PlayerModule, lastSideIdx, laneIdx, false, true)
            end
            heldSustains[laneIdx] = nil
        end
        vimRelease(laneIdx)
    end
    vibratoTimers = {}
end

local function findLocalPlayerSlot(players)
    local ov = settings.sideOverride
    if ov == 1 or ov == 2 then
        local p = players[ov]
        if type(p) == "table" and p.Strums then return ov, p end
    end
    for k, p in pairs(players) do
        if type(p) == "table" and p.Strums and p.Controller ~= nil and not p.Ghost then
            return k, p
        end
    end
    for k, p in pairs(players) do
        if type(p) == "table" and p.Strums and not p.Ghost then
            return k, p
        end
    end
    return nil, nil
end

local function planFor(note)
    local plan = plannedDecision[note]
    if plan then return plan end

    plan = { jitter = 0, releaseJ = 0, miss = false, bad = false, mine = false, mustHit = false }

    if settings.skipMines and note.mint and mintIsDamaging[note.mint] then
        plan.mine = true
        plannedDecision[note] = plan
        return plan
    end

    if note.mint and mintIsMustHit[note.mint] then
        plan.mustHit = true
        plannedDecision[note] = plan
        return plan
    end

    plan.jitter   = jitterMs(settings.pressJitterMs,   settings.pressJitterClampMs)
    plan.releaseJ = jitterMs(settings.releaseJitterMs, settings.releaseJitterClampMs)

    if not (note.len and note.len > 0) then
        local mc, bc = settings.missChance, settings.badChance
        if mc > 0 or bc > 0 then
            local r = rng:NextNumber() * 100
            if r < mc then
                plan.miss = true
            elseif r < mc + bc then
                local sign = (rng:NextNumber() < 0.5) and -1 or 1
                plan.jitter = sign * (80 + rng:NextNumber() * 50)
                plan.bad = true
            end
        end
    end

    plannedDecision[note] = plan
    return plan
end

-- FEATURE 2: Accuracy Guard check
------------------------------------------------------------
local function checkAccuracyGuard()
    if not settings.accuracyGuard then return end
    local total = stats.notesHit + stats.notesMissed + stats.notesBad
    if total < 10 then return end  -- need enough samples
    local acc = getAccuracy()
    if acc < settings.accuracyThreshold then
        settings.enabled = false
        releaseAllPressed()
        clearState()
        hadPlayer = false
        if notify then
            notify({
                Title    = "Accuracy Guard triggered",
                Content  = string.format("Accuracy %.1f%% < threshold %.1f%%. Autoplay disabled.", acc, settings.accuracyThreshold),
                Icon     = "shield-off",
                Duration = 6,
            })
        end
    end
end

-- FEATURE 14: Timing calibration — rolling average
------------------------------------------------------------
local MAX_DELTA_SAMPLES = 50

local function recordDelta(deltaMs)
    table.insert(stats.timingDeltas, deltaMs)
    if #stats.timingDeltas > MAX_DELTA_SAMPLES then
        table.remove(stats.timingDeltas, 1)
    end
    local sum = 0
    for _, v in ipairs(stats.timingDeltas) do sum = sum + v end
    stats.avgDeltaMs = sum / #stats.timingDeltas
end

------------------------------------------------------------
-- FEATURE 11: Note density (rolling 1-second window)
------------------------------------------------------------
local function recordHit(now)
    table.insert(stats.recentHitTimes, now)
    -- prune entries older than 1 second
    local cutoff = now - 1.0
    local i = 1
    while i <= #stats.recentHitTimes do
        if stats.recentHitTimes[i] < cutoff then
            table.remove(stats.recentHitTimes, i)
        else
            i = i + 1
        end
    end
    stats.notesPerSec = #stats.recentHitTimes
end

local lastChordTick  = -1
local chordLanesThisTick = 0

local function recordChordLane(now)
    if mathAbs(now - lastChordTick) < 0.005 then
        chordLanesThisTick = chordLanesThisTick + 1
        if chordLanesThisTick >= 2 then
            stats.chords = stats.chords + 1

    end
end

-- FEATURE 3 / 12 / 13: HUD scraper (song name, HP, score)
------------------------------------------------------------
local hudCache         = nil
local lastHudScrape    = 0
local HUD_SCRAPE_RATE  = 1.0

local function scrapeHud()
    local now = os.clock()
    if now - lastHudScrape < HUD_SCRAPE_RATE then return end
    lastHudScrape = now

    local plr = Players.LocalPlayer
    local pg  = plr and plr:FindFirstChildOfClass("PlayerGui")
    if not pg then return end
    local hud = pg:FindFirstChild("GameHUD")
    if not hud then hudCache = nil; return end
    if hudCache ~= hud then hudCache = hud end

    -- Song name
    pcallF(function()
        local label = hud:FindFirstChild("SongName", true)
                   or hud:FindFirstChild("Song",     true)
                   or hud:FindFirstChild("Title",    true)
        if label and label:IsA("TextLabel") and #label.Text > 0 then
            stats.songName = label.Text
        end
    end)

    -- Score
    pcallF(function()
        local label = hud:FindFirstChild("Score",     true)
                   or hud:FindFirstChild("ScoreText", true)
                   or hud:FindFirstChild("Points",    true)
        if label and label:IsA("TextLabel") then
            stats.currentScore = label.Text
        end
    end)

    -- Health / HP
    pcallF(function()
        local bar = hud:FindFirstChild("HealthBar", true)
                 or hud:FindFirstChild("HPBar",     true)
                 or hud:FindFirstChild("Health",    true)
        if bar then
            -- Try numeric value first
            if bar:IsA("TextLabel") then
                stats.currentHealth = bar.Text
                local hp = tonumber(bar.Text:match("%d+"))
                if hp and settings.healthMonitor and hp <= settings.healthPauseThreshold then
                    -- FEATURE 12: critical HP warning
                    if notify then
                        notify({ Title = "⚠ Low HP!", Content = "Health at " .. tostring(hp) .. " — consider pausing.", Icon = "heart", Duration = 4 })
                    end
                end
            elseif bar:IsA("Frame") and bar:FindFirstChild("Bar") then
                local inner = bar:FindFirstChild("Bar")
                local pct   = math.floor(inner.Size.X.Scale * 100)
                stats.currentHealth = pct .. "%"
                if settings.healthMonitor and pct <= settings.healthPauseThreshold then
                    if notify then
                        notify({ Title = "⚠ Low HP!", Content = "Health at " .. pct .. "% — consider pausing.", Icon = "heart", Duration = 4 })
                    end
                end
            end
        end
    end)
end

local function doVibratoTick(now)
    if not settings.sustainVibrato then return end
    local ivSec = settings.vibratoIntervalMs / 1000
    for laneIdx, note in pairs(heldSustains) do
        local last = vibratoTimers[laneIdx] or 0
        if now - last >= ivSec then
            vibratoTimers[laneIdx] = now
            -- micro re-trigger: release then press again via VIM only (not Trigger)
            local kc = keyCodeForLane(laneIdx)
            if kc then
                pcallF(VIM.SendKeyEvent, VIM, false, kc, false, game)
                task.delay(0.012, function()
                    pcallF(VIM.SendKeyEvent, VIM, true, kc, false, game)
                end)
            end
        end
    end
end

-- FEATURE 16: Ghost Input

local lastGhostTime = 0

local function ghostInputTick(now)
    if not settings.ghostInputMode then return end
    if now - lastGhostTime < 0.3 then return end
    if rng:NextNumber() > settings.ghostChance * 300 then return end
    lastGhostTime = now
    -- Pick a lane that is NOT currently held
    local candidates = {}
    for i = 1, detectedLaneCount do
        if not heldSustains[i] then candidates[#candidates+1] = i end
    end
    if #candidates == 0 then return end
    local lane = candidates[math.random(1, #candidates)]
    local kc   = keyCodeForLane(lane)
    if kc then
        pcallF(VIM.SendKeyEvent, VIM, true,  kc, false, game)
        task.delay(0.01 + rng:NextNumber() * 0.02, function()
            pcallF(VIM.SendKeyEvent, VIM, false, kc, false, game)
        end)
    end
end

-- Main autoplay tick
------------------------------------------------------------
local function autoplayTick()
    if not settings.enabled then return end

    local players = getPlayers()
    if not players then
        if hadPlayer then releaseAllPressed(); hadPlayer = false end
        return
    end

    local sideIdx, p = findLocalPlayerSlot(players)
    if not p then
        if hadPlayer then releaseAllPressed(); hadPlayer = false end
        return
    end
    hadPlayer   = true
    lastSideIdx = sideIdx

    -- FEATURE 18: auto-release on pause
    local isPaused = p.BlockPlay or p.Killed or false
    if isPaused then
        if not wasPaused and settings.autoReleaseOnPause then
            releaseAllPressed()
            wasPaused = true
        end
        return
    end
    wasPaused = false

    local hitTime    = getCurrentTime() - getInputOffset()
    local pressOff   = settings.pressOffsetMs
    local releaseOff = settings.releaseOffsetMs
    local leeway     = settings.pressLeewayMs
    local strums     = p.Strums
    local now        = os.clock()

    updateLaneCount(strums)

    -- Song change detection
    local totalNotes = 0
    for laneIdx = 1, #strums do
        local s = strums[laneIdx]
        if s and s.Notes then totalNotes = totalNotes + #s.Notes end
    end
    if lastNoteCount > 20 and totalNotes < 5 then
        resetStats(true)
        clearState()
        fcTriggered = false  -- reset FC guard for new song
    end
    lastNoteCount = totalNotes

    -- FEATURE 19: vibrato on held sustains
    doVibratoTick(now)

    -- FEATURE 16: ghost inputs
    ghostInputTick(now)

    for laneIdx = 1, #strums do
        local strum = strums[laneIdx]

        -- Sustain release
        local heldNote = heldSustains[laneIdx]
        if heldNote then
            local plan      = plannedDecision[heldNote] or planFor(heldNote)
            local releaseAt = heldNote.t + (heldNote.len or 0) + releaseOff + plan.releaseJ
            if heldNote.removed or hitTime >= releaseAt then
                pcallF(Trigger, PlayerModule, sideIdx, laneIdx, false, true)
                vimRelease(laneIdx)
                heldSustains[laneIdx]  = nil
                vibratoTimers[laneIdx] = nil
            end
        end

        -- Press
        local closestIdx = strum.Closest
        local notes      = strum.Notes
        local note       = closestIdx and notes and notes[closestIdx]
        if note and not pressedNotes[note] then
            local plan = planFor(note)

            if plan.mine then
                pressedNotes[note]   = true
                stats.minesSkipped   = stats.minesSkipped + 1
                stats.streak         = 0

            elseif plan.miss then
                pressedNotes[note]  = true
                stats.notesMissed   = stats.notesMissed + 1
                stats.streak        = 0
                -- FEATURE 1: FC Guard
                if settings.fcGuard then fcGuardDisable("miss planned") end
                -- FEATURE 20: miss log
                if settings.missLog then
                    table.insert(stats.missPositions, { lane = laneIdx, t = note.t, time = hitTime })
                    if settings.debugMode then
                        print(string.format("[RAT MISS] Lane %d @ song-t %.3f | hit-t %.3f", laneIdx, note.t, hitTime))
                    end
                end
                -- FEATURE 2: accuracy guard
                checkAccuracyGuard()

            else
                -- FEATURE 4: lane muting
                if settings.mutedLanes[laneIdx] then
                    pressedNotes[note] = true
                else
                    local target    = note.t + pressOff + plan.jitter
                    local timeDiff  = hitTime - target

                    -- FEATURE 8: late hit safety net — extra leeway past the ideal window
                    local effectiveLeeway = leeway
                    if timeDiff > 0 and settings.lateHitSafety then
                        effectiveLeeway = effectiveLeeway + settings.lateHitExtraMs
                    end

                    if timeDiff >= -effectiveLeeway then
                        pressedNotes[note] = true

                        -- Stats
                        if plan.bad then
                            stats.notesBad = stats.notesBad + 1
                            stats.streak   = 0
                            if settings.fcGuard then fcGuardDisable("bad note") end
                        else
                            stats.notesHit = stats.notesHit + 1
                            if plan.mustHit then stats.mustHitsHit = stats.mustHitsHit + 1 end
                            stats.streak = stats.streak + 1
                            if stats.streak > stats.bestStreak then
                                stats.bestStreak = stats.streak
                            end
                        end

                        -- FEATURE 9: chord detection
                        recordChordLane(now)

                        -- FEATURE 11: note density
                        recordHit(now)

                        -- FEATURE 14: timing calibration (store delta in ms)
                        local deltaMs = (hitTime - note.t) * 1000
                        recordDelta(deltaMs)

                        -- FEATURE 2: accuracy guard
                        checkAccuracyGuard()

                        -- Debug
                        if settings.debugMode then
                            print(string.format("[RAT] Lane %d | Δ %.2f ms | jitter %.2f ms | %s",
                                laneIdx, deltaMs, plan.jitter,
                                plan.mustHit and "MUSTHIT" or (plan.bad and "BAD" or "HIT")))
                        end

                        vimPress(laneIdx)
                        pcallF(Trigger, PlayerModule, sideIdx, laneIdx, true, true)
                        if note.len and note.len > 0 then
                            heldSustains[laneIdx]  = note
                            vibratoTimers[laneIdx] = now
                        else
                            local capturedLane = laneIdx
                            taskDefer(function()
                                pcallF(Trigger, PlayerModule, sideIdx, capturedLane, false, true)
                                vimRelease(capturedLane)
                            end)
                        end
                    end
                end
            end
        end
    end
end

local botplayLabel  = nil

local function refreshBotplayLabel()
    botplayLabel = nil
    local plr = Players.LocalPlayer
    local pg  = plr and plr:FindFirstChildOfClass("PlayerGui")
    if not pg then return end
    local hud = pg:FindFirstChild("GameHUD")
    if not hud then return end
    for _, d in ipairs(hud:GetDescendants()) do
        if d.Name == "Botplay" and d:IsA("GuiObject") then
            botplayLabel = d
            return
        end
    end
end

local function spoofBotplayLabel()
    if not settings.hideBotplayLabel then return end
    local now = os.clock()
    if now - lastSpoofTime < 0.5 then return end
    lastSpoofTime = now
    if not botplayLabel or not botplayLabel.Parent then refreshBotplayLabel() end
    if botplayLabel and botplayLabel.Visible then botplayLabel.Visible = false end
end

-- 12) Master heartbeat — all subsystems in one connection
------------------------------------------------------------
local lastStatUpdate = 0

local masterConn = RunService.Heartbeat:Connect(function()
    local now = os.clock()

    autoplayTick()
    spoofBotplayLabel()
    antiAfkTick()
    scrapeHud()

    -- Stats label refresh (~4 Hz)
    if now - lastStatUpdate >= 0.25 then
        lastStatUpdate = now

        local acc    = string.format("%.2f%%", getAccuracy())
        local uptime = mathFloor(now - stats.sessionStart)
        local mins   = mathFloor(uptime / 60)
        local secs   = uptime % 60

        -- Which lanes are held right now?
        local heldStr = ""
        for i = 1, detectedLaneCount do
            if heldSustains[i] then heldStr = heldStr .. i .. " " end
        end
        if heldStr == "" then heldStr = "none" end

        safeSetDesc(pSong,     stats.songName)
        safeSetDesc(pScore,    stats.currentScore)
        safeSetDesc(pHealth,   stats.currentHealth)
        safeSetDesc(pAccuracy, acc)
        safeSetDesc(pHits,     tostring(stats.notesHit) .. (stats.mustHitsHit > 0 and ("  (must-hits: " .. stats.mustHitsHit .. ")") or ""))
        safeSetDesc(pBads,     "Bad: " .. stats.notesBad .. "  |  Missed: " .. stats.notesMissed)
        safeSetDesc(pMines,    tostring(stats.minesSkipped))
        safeSetDesc(pChords,   tostring(stats.chords))
        safeSetDesc(pStreak,   stats.streak .. "  /  best " .. stats.bestStreak)
        safeSetDesc(pDensity,  string.format("%.1f / sec", stats.notesPerSec))
        safeSetDesc(pCalib,    string.format("%.2f ms  (samples: %d)", stats.avgDeltaMs, #stats.timingDeltas))
        safeSetDesc(pSustains, heldStr)
        safeSetDesc(pSongCount, tostring(stats.songCount))
        safeSetDesc(pUptime,   string.format("%d m %02d s", mins, secs))

        -- History tab
        for i, entry in ipairs(stats.songHistory) do
            safeSetDesc(histParagraphs[i],
                string.format("%s | %s | Hits: %d | Miss: %d | Bad: %d | Streak: %d",
                    entry.name, entry.acc, entry.hits, entry.misses, entry.bads, entry.best))
        end
    end
end)

-- 13) FEATURE 17: Quick toggle hotkey via UserInputService
------------------------------------------------------------
local quickKeyConn = UIS.InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed then return end
    if not Enum.KeyCode[settings.quickKey] then return end
    if input.KeyCode == Enum.KeyCode[settings.quickKey] then
        settings.enabled = not settings.enabled
        if not settings.enabled then
            releaseAllPressed()
            clearState()
            hadPlayer   = false
            fcTriggered = false
        end
        notify({
            Title   = settings.enabled and "Autoplay ON" or "Autoplay OFF",
            Content = "Quick toggled via " .. settings.quickKey,
            Icon    = settings.enabled and "check" or "pause",
            Duration = 2,
        })
    end
end)
