do
    local function killState(g)
        if type(g) ~= "table" then return end
        pcall(function() g.heartbeat:Disconnect() end)
        pcall(function() g.spoof:Disconnect() end)
        pcall(function() g.afk:Disconnect() end)

            for _, p in pairs(g.players) do

            end
        end
    end
    killState(_G.__RAT_AUTOPLAY); _G.__RAT_AUTOPLAY = nil
    killState(_G.__FF_AUTOPLAY);  _G.__FF_AUTOPLAY  = nil
    if _G.__AUTOPLAY_CONN       then pcall(function() _G.__AUTOPLAY_CONN:Disconnect() end);       _G.__AUTOPLAY_CONN       = nil end
    if _G.__AUTOPLAY_SPOOF_CONN then pcall(function() _G.__AUTOPLAY_SPOOF_CONN:Disconnect() end); _G.__AUTOPLAY_SPOOF_CONN = nil end
    pcall(game:GetService("RunService").UnbindFromRenderStep,

end

-- 6) Auto-detect bound keys from in-game preferences (up to 7 lanes)
------------------------------------------------------------
local function pickKeyboardKey(bind)
    if type(bind) ~= "table" then return nil end
    local fallback
    for _, v in pairs(bind) do
        local name
        if type(v) == "string" then name = v
        elseif typeof(v) == "EnumItem" then name = v.Name end
        if name then
            local up = name:upper()
            if Enum.KeyCode[up] then
                if up ~= "LEFT" and up ~= "RIGHT" and up ~= "UP" and up ~= "DOWN" then
                    return up
                end
                fallback = fallback or up
            end
        end
    end
    return fallback
end

do
    local plr = Players.LocalPlayer
    local getPref = plr and plr:FindFirstChild("GetPref")
    if getPref then
        local ok, binds = pcallF(function() return getPref:Invoke("Binds") end)
        if ok and type(binds) == "table" then
            local laneNames = { "Left", "Down", "Up", "Right", "Extra1", "Extra2", "Extra3" }
            for i, n in ipairs(laneNames) do
                local k = pickKeyboardKey(binds[n])
                if k then settings.bindKey[i] = k end
            end
        end
    end
end

------------------------------------------------------------
-- 7) RNG / Gauss (Box-Muller)
------------------------------------------------------------
local rng = Random.new(tick() * 1e3)

local function gauss()
    local u1 = mathMax(rng:NextNumber(), 1e-7)
    return mathSqrt(-2 * mathLog(u1)) * mathCos(2 * mathPi * rng:NextNumber())
end

local function jitterMs(sigma, clampMag)
    if sigma <= 0 then return 0 end
    local v = gauss() * sigma
    if v < -clampMag then return -clampMag end
    if v >  clampMag then return  clampMag end
    return v
end

local function keyCodeForLane(laneIdx)
    if not settings.sendVirtualInput then return nil end
    local name = settings.bindKey[laneIdx]
    if not name or name == "" then return nil end
    return Enum.KeyCode[name:upper()]

local function vimPress(lane)

    local kc = keyCodeForLane(lane)

end

-- 9) Engine state
------------------------------------------------------------
local pressedNotes    = setmetatable({}, { __mode = "k" })
local heldSustains    = {}
local plannedDecision = setmetatable({}, { __mode = "k" })

local function clearState()
    pressedNotes    = setmetatable({}, { __mode = "k" })
    plannedDecision = setmetatable({}, { __mode = "k" })
    heldSustains    = {}
end

local function findLocalPlayerSlot(players)
    local ov = settings.sideOverride
    if ov == 1 or ov == 2 then
        local p = players[ov]
        if type(p) == "table" and p.Strums then return ov, p end
    end
    for k, p in pairs(players) do
        if type(p) == "table" and p.Strums and p.Controller ~= nil and not p.Ghost then
            return k, p
        end
    end
    for k, p in pairs(players) do
        if type(p) == "table" and p.Strums and not p.Ghost then
            return k, p
        end
    end
    return nil, nil
end

local function planFor(note)
    local plan = plannedDecision[note]
    if plan then return plan end

    plan = { jitter = 0, releaseJ = 0, miss = false, bad = false, mine = false, mustHit = false }

    if settings.skipMines and note.mint and mintIsDamaging[note.mint] then
        plan.mine = true
        plannedDecision[note] = plan
        return plan
    end

    if note.mint and mintIsMustHit[note.mint] then
        plan.mustHit = true
        plannedDecision[note] = plan
        return plan
    end

    plan.jitter   = jitterMs(settings.pressJitterMs,   settings.pressJitterClampMs)
    plan.releaseJ = jitterMs(settings.releaseJitterMs, settings.releaseJitterClampMs)

    if not (note.len and note.len > 0) then
        local mc, bc = settings.missChance, settings.badChance
        if mc > 0 or bc > 0 then
            local r = rng:NextNumber() * 100
            if r < mc then
                plan.miss = true
            elseif r < mc + bc then
                local sign = (rng:NextNumber() < 0.5) and -1 or 1
                plan.jitter = sign * (80 + rng:NextNumber() * 50)
                plan.bad = true
            end
        end
    end

    plannedDecision[note] = plan
    return plan
end

-- 9b) Auto-detect lane count from current player
------------------------------------------------------------
local detectedLaneCount = 4

local function updateLaneCount(strums)
    local n = #strums
    if n > detectedLaneCount then
        detectedLaneCount = n
    end
end

------------------------------------------------------------
-- Song change detection (reset stats when strum count resets)
------------------------------------------------------------
local lastNoteCount = 0

local hadPlayer   = false
local lastSideIdx = nil

local function releaseAllPressed(maxLanes)
    maxLanes = maxLanes or detectedLaneCount
    for laneIdx = 1, maxLanes do
        if heldSustains[laneIdx] then
            if lastSideIdx then
                pcallF(Trigger, PlayerModule, lastSideIdx, laneIdx, false, true)
            end
            heldSustains[laneIdx] = nil
        end
        vimRelease(laneIdx)
    end
end

local function autoplayTick()
    if not settings.enabled then return end

    local players = getPlayers()
    if not players then
        if hadPlayer then
            releaseAllPressed()
            hadPlayer = false
        end
        return
    end

    local sideIdx, p = findLocalPlayerSlot(players)
    if not p then
        if hadPlayer then
            releaseAllPressed()
            hadPlayer = false
        end
        return
    end
    hadPlayer   = true
    lastSideIdx = sideIdx
    if p.BlockPlay or p.Killed then return end

    local hitTime    = getCurrentTime() - getInputOffset()
    local pressOff   = settings.pressOffsetMs
    local releaseOff = settings.releaseOffsetMs
    local leeway     = settings.pressLeewayMs
    local strums     = p.Strums

    updateLaneCount(strums)

    -- Detect song restart (total note count drops drastically)
    local totalNotes = 0
    for laneIdx = 1, #strums do
        local s = strums[laneIdx]
        if s and s.Notes then totalNotes = totalNotes + #s.Notes end
    end
    if lastNoteCount > 20 and totalNotes < 5 then
        resetStats()
        clearState()
    end
    lastNoteCount = totalNotes

    for laneIdx = 1, #strums do
        local strum = strums[laneIdx]

        -- Sustain release
        local heldNote = heldSustains[laneIdx]
        if heldNote then
            local plan    = plannedDecision[heldNote] or planFor(heldNote)
            local releaseAt = heldNote.t + (heldNote.len or 0) + releaseOff + plan.releaseJ
            if heldNote.removed or hitTime >= releaseAt then
                pcallF(Trigger, PlayerModule, sideIdx, laneIdx, false, true)
                vimRelease(laneIdx)
                heldSustains[laneIdx] = nil
            end
        end

        -- Press
        local closestIdx = strum.Closest
        local notes      = strum.Notes
        local note       = closestIdx and notes and notes[closestIdx]
        if note and not pressedNotes[note] then
            local plan = planFor(note)
            if plan.mine then
                -- Skip damaging notes
                pressedNotes[note] = true
                stats.minesSkipped = stats.minesSkipped + 1
                stats.streak = 0
            elseif plan.miss then
                pressedNotes[note] = true
                stats.notesMissed  = stats.notesMissed + 1
                stats.streak       = 0
            else
                local target = note.t + pressOff + plan.jitter
                if hitTime - target >= -leeway then
                    pressedNotes[note] = true

                    -- Stats tracking
                    if plan.bad then
                        stats.notesBad = stats.notesBad + 1
                        stats.streak   = 0
                    else
                        stats.notesHit = stats.notesHit + 1
                        if plan.mustHit then
                            stats.mustHitsHit = stats.mustHitsHit + 1
                        end
                        stats.streak = stats.streak + 1
                        if stats.streak > stats.bestStreak then
                            stats.bestStreak = stats.streak
                        end
                    end

                    -- Debug timing
                    if settings.debugMode then
                        local delta = (hitTime - note.t) * 1000
                        print(string.format("[RAT] Lane %d | Δ %.2f ms | jitter %.2f ms | %s",
                            laneIdx, delta, plan.jitter,
                            plan.mustHit and "MUSTHIT" or (plan.bad and "BAD" or "HIT")))
                    end

                    vimPress(laneIdx)
                    pcallF(Trigger, PlayerModule, sideIdx, laneIdx, true, true)
                    if note.len and note.len > 0 then
                        heldSustains[laneIdx] = note
                    else
                        local capturedLane = laneIdx
                        taskDefer(function()
                            pcallF(Trigger, PlayerModule, sideIdx, capturedLane, false, true)
                            vimRelease(capturedLane)
                        end)
                    end
                end
            end
        end
    end
end

-- 10) Botplay label spoof — cached + throttled
------------------------------------------------------------
local botplayLabel  = nil
local lastSpoofTime = 0

local function refreshBotplayLabel()
    botplayLabel = nil
    local plr = Players.LocalPlayer
    local pg  = plr and plr:FindFirstChildOfClass("PlayerGui")
    if not pg then return end
    local hud = pg:FindFirstChild("GameHUD")
    if not hud then return end
    for _, d in ipairs(hud:GetDescendants()) do
        if d.Name == "Botplay" and d:IsA("GuiObject") then
            botplayLabel = d
            return
        end
    end
end

local function spoofBotplayLabel()
    if not settings.hideBotplayLabel then return end
    local now = os.clock()
    if now - lastSpoofTime < 0.5 then return end
    lastSpoofTime = now
    if not botplayLabel or not botplayLabel.Parent then
        refreshBotplayLabel()
    end
    if botplayLabel and botplayLabel.Visible then
        botplayLabel.Visible = false
    end
end

local lastAfkTime = os.clock()

local function antiAfkTick()
    if not settings.antiAfk then return end
    local now = os.clock()
    if now - lastAfkTime < settings.antiAfkIntervalSec then return end
    lastAfkTime = now
    -- Tiny virtual input burst (space down/up) that is invisible in-game
    pcallF(VIM.SendKeyEvent, VIM, true,  Enum.KeyCode.Space, false, game)
    pcallF(VIM.SendKeyEvent, VIM, false, Enum.KeyCode.Space, false, game)
end

local function makeSlider(tab, opts)
    local function fire(v)
        opts.Callback(v)
        if opts.DangerBelow and v < opts.DangerBelow then
            notify({
                Title    = "Risky setting",
                Content  = (opts.Title or "value") .. " = " .. tostring(v) ..
                           " is below the safe threshold (" .. tostring(opts.DangerBelow) ..
                           "). This may trigger the anti-cheat.",
                Icon     = "alert-triangle",
                Duration = 5,
            })
        end
    end
    if tab.Slider then
        return tab:Slider({
            Title    = opts.Title,
            Icon     = opts.Icon,
            Value    = { Min = opts.Min, Max = opts.Max, Default = opts.Default },
            Step     = opts.Step or 1,
            Callback = fire,
        })
    end
    return tab:Input({
        Title       = opts.Title .. " (" .. tostring(opts.Min) .. ".." .. tostring(opts.Max) .. ")",
        Value       = tostring(opts.Default),
        InputIcon   = opts.Icon or "hash",
        Placeholder = "number",
        Callback    = function(text)
            local n = tonumber(text)
            if not n then return end
            if opts.Min and n < opts.Min then n = opts.Min end
            if opts.Max and n > opts.Max then n = opts.Max end
            fire(n)
        end,
    })
end

local function safeSetDesc(handle, text)
    if not handle then return end
    pcallF(function()
        if handle.SetDesc then
            handle:SetDesc(text)
        elseif handle.Paragraph then
            handle.Paragraph.Desc.Text = text
        end
    end)
end

local lastStatUpdate = 0
local function updateStatLabels()
    local now = os.clock()
    if now - lastStatUpdate < 0.25 then return end
    lastStatUpdate = now

    local acc    = string.format("%.2f%%", getAccuracy())
    local uptime = math.floor(now - stats.sessionStart)
    local mins   = math.floor(uptime / 60)
    local secs   = uptime % 60

    safeSetDesc(pAccuracy,  acc)
    safeSetDesc(pHits,      tostring(stats.notesHit) ..
                            (stats.mustHitsHit > 0 and ("  (must-hits: " .. stats.mustHitsHit .. ")") or ""))
    safeSetDesc(pBads,      "Bad: " .. stats.notesBad .. "  |  Missed: " .. stats.notesMissed)
    safeSetDesc(pMines,     tostring(stats.minesSkipped))
    safeSetDesc(pStreak,    stats.streak .. "  /  best " .. stats.bestStreak)
    safeSetDesc(pSongCount, tostring(stats.songCount))
    safeSetDesc(pUptime,    string.format("%d m %02d s", mins, secs))
end

    InputIcon   = "key",

        if not txt or #txt == 0 then return end
        local up = txt:sub(1,1):upper() .. txt:sub(2)
        if Enum.KeyCode[up] then
            settings.toggleKey = up
            Window:SetToggleKey(Enum.KeyCode[up])
            notify({ Title = "Toggle key changed", Content = "Menu now opens with: " .. up, Icon = "key", Duration = 3 })
        else
            notify({ Title = "Invalid key", Content = "\"" .. txt .. "\" is not a valid KeyCode name.", Icon = "alert-triangle", Duration = 3 })
        end
    end,
})
