local function getChar()   return LocalPlayer.Character end
local function getHRP()    local c=getChar() return c and c:FindFirstChild('HumanoidRootPart') end
local function getHum()    local c=getChar() return c and c:FindFirstChildOfClass('Humanoid') end
local function getTool()   local c=getChar() return c and c:FindFirstChildOfClass('Tool') end
local function getHandle() local t=getTool() return t and (t:FindFirstChild('Handle') or t:FindFirstChildWhichIsA('BasePart')) end

local function getClosest(maxD)
    maxD = maxD or Cfg.maxRange
    local hrp = getHRP()
    if not hrp then return nil end
    local best, bestD = nil, maxD
    for _,p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character then
            local ph = p.Character:FindFirstChild('HumanoidRootPart')
            if ph then
                local d = (hrp.Position - ph.Position).Magnitude
                if d < bestD then
                    local hum = p.Character:FindFirstChildOfClass('Humanoid')
                    if hum and hum.Health > 0 then best=p bestD=d end
                end
            end
        end
    end
    return best
end

local function fireTouchAll(handle, char)
    for _,p in ipairs(char:GetDescendants()) do
        if p:IsA('BasePart') and p.Parent then
            firetouchinterest(handle, p, 0)
            firetouchinterest(handle, p, 1)
        end
    end
end

-- Aimbot heartbeat
local killThrottle = 0
RunService.Heartbeat:Connect(function()
    local hrp = getHRP()
    if not hrp then return end
    local myPos = hrp.Position
    local best, bestD, bestPart = nil, Cfg.maxRange, nil
    for _,p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character then
            local ph = p.Character:FindFirstChild('HumanoidRootPart')
            if ph then
                local d=(ph.Position-myPos).Magnitude
                if d<=Cfg.maxRange then
                    local hum=p.Character:FindFirstChildOfClass('Humanoid')
                    if hum and hum.Health>0 and d<bestD then
                        best=p bestD=d bestPart=ph
                    end
                end
            end
        end
    end
    if bestPart then
        HL.CFrame=CFrame.new(bestPart.Position)
        if Cfg.AimbotEnabled then
            local dir=Vector3.new((bestPart.Position-myPos).X,0,(bestPart.Position-myPos).Z).Unit
            if Cfg.Wiggle then
                local wx = (Cfg.WiggleMode=='Left' or Cfg.WiggleMode=='Both') and math.random(-1,1) or 0
                local wz = (Cfg.WiggleMode=='Right' or Cfg.WiggleMode=='Both') and math.random(-1,1) or 0
                dir=Vector3.new(dir.X+wx,0,dir.Z+wz).Unit
            end
            hrp.CFrame=CFrame.new(myPos)*CFrame.lookAt(Vector3.new(),dir)
            if Cfg.AutoSwing then local t=getTool() if t then t:Activate() end end
            -- Teleport strafe
            if UserInputService:IsKeyDown(Enum.KeyCode.T) then
                local d2=Cfg.TPDistance+(Cfg.randomizeTeleport and math.random(-2,2) or 0)
                local tp=bestPart.Position-bestPart.CFrame.LookVector*d2
                local td=Vector3.new((bestPart.Position-tp).X,0,(bestPart.Position-tp).Z).Unit
                hrp.CFrame=CFrame.new(tp)*CFrame.lookAt(Vector3.new(),td)
            end
        end
    else
        HL.CFrame=CFrame.new(0,-10000,0)
    end
    -- Kill Aura
    if Cfg.KillAuraEnabled then
        local now=tick()
        if (now-killThrottle)>=0.08 then
            killThrottle=now
            local h=getHandle()
            if h then
                for _,p in ipairs(Players:GetPlayers()) do
                    if p~=LocalPlayer and p.Character then
                        local ph=p.Character:FindFirstChild('HumanoidRootPart')
                        if ph and (hrp.Position-ph.Position).Magnitude<=Cfg.maxRange then
                            local hum=p.Character:FindFirstChildOfClass('Humanoid')
                            if hum and hum.Health>0 then
                                local ch=p.Character
                                task.spawn(function() fireTouchAll(h,ch) end)
                                local t=getTool() if t then t:Activate() end
                            end
                        end
                    end
                end
            end
        end
    end
    -- Auto Walk
    if Cfg.AimbotEnabled and Cfg.AutoWalk then
        local hum=getHum()
        if hum then
            if best and best.Character then
                local ph=best.Character:FindFirstChild('HumanoidRootPart')
                if ph then hum:MoveTo(ph.Position) end
            else hum:MoveTo(hrp.Position) end
        end
    end
    -- WalkSpeed bypass
    if Cfg.WalkBypass and Cfg.TargetWalkspeed>16 then
        local hum=getHum()
        if hum and hum.MoveDirection.Magnitude>0 then
            hrp.CFrame=hrp.CFrame*CFrame.new(0,0,-0.135)
        end
    end
end)

-- RenderStepped loop (body lock, spin, jump, autohit, flick)
RunService.RenderStepped:Connect(function()
    local hrp=getHRP()
    local hum=getHum()
    if not (hrp and hum) then return end
    local tgt=getClosest()
    local tchar=tgt and tgt.Character
    local tpart=tchar and tchar:FindFirstChild(Cfg.LockBodyPart)
    -- Body lock
    if Cfg.BodyLock and tpart then
        local aim=tpart.Position
        if Cfg.Prediction then
            local th=tchar:FindFirstChild('HumanoidRootPart')
            if th then aim=aim+th.AssemblyLinearVelocity*Cfg.PredictionOffset end
        end
        hrp.CFrame=CFrame.new(hrp.Position,Vector3.new(aim.X,hrp.Position.Y,aim.Z))
        hum.AutoRotate=false
    else hum.AutoRotate=true end
    -- Auto hit
    if Cfg.AutoHit and tpart and (hrp.Position-tpart.Position).Magnitude<=Cfg.AutoHitStuds then
        local t=getTool() if t and t:FindFirstChild('Handle') then t:Activate() end
    end
    -- Spin
    if Cfg.SpinEnabled then hrp.CFrame=hrp.CFrame*CFrame.Angles(0,math.rad(Cfg.SpinSpeed),0) end
    -- Auto jump
    if Cfg.AutoJump and tpart and (hrp.Position-tpart.Position).Magnitude<=Cfg.JumpStuds then
        if hum.FloorMaterial~=Enum.Material.Air then hum:ChangeState(Enum.HumanoidStateType.Jumping) end
    end
end)

-- Auto Snipe
local snipeTarget=nil
task.spawn(function()
    while task.wait(0) do
        if Cfg.AutoSnipe then
            local hrp=getHRP() if not hrp then task.wait(0.3) continue end
            local best,bestT=nil,math.huge
            for _,p in ipairs(Players:GetPlayers()) do
                if p~=LocalPlayer and p.Character then
                    local ls=p:FindFirstChild('leaderstats')
                    local t=ls and ls:FindFirstChild('Time') and ls.Time.Value or 0
                    local ph=p.Character:FindFirstChild('HumanoidRootPart')
                    local hum=p.Character:FindFirstChildOfClass('Humanoid')
                    if ph and hum and hum.Health>0 and t>=Cfg.SnipeMinTime then
                        if t<bestT then best=p bestT=t end
                    end
                end
            end
            if best and best~=snipeTarget then
                snipeTarget=best
                task.spawn(function()
                    while snipeTarget and Cfg.AutoSnipe do
                        local tc=snipeTarget.Character
                        local th=tc and tc:FindFirstChildOfClass('Humanoid')
                        if not(tc and th and th.Health>0) then snipeTarget=nil break end
                        local hrp2=getHRP() local ph=tc:FindFirstChild('HumanoidRootPart')
                        if hrp2 and ph then
                            hrp2.CFrame=CFrame.new(ph.Position+Vector3.new(0,0,3))
                            local tool=LocalPlayer.Backpack:FindFirstChildOfClass('Tool')
                            if tool then
                                tool.Parent=getChar()
                                task.wait() if tool then tool:Activate() end
                            end
                        end
                        task.wait(0.05)
                    end
                end)
            end
            if snipeTarget then
                local tc=snipeTarget.Character
                local th=tc and tc:FindFirstChildOfClass('Humanoid')
                if not(tc and th and th.Health>0) then snipeTarget=nil end
            end
        end
        task.wait(0)
    end
end)

-- Server Hop
local function serverHop(sizeType)
    local ok,data=pcall(function()
        return HttpService:JSONDecode(game:HttpGet('https://games.roblox.com/v1/games/'..game.PlaceId..'/servers/Public?sortOrder=Desc&limit=100'))
    end)
    if ok and data and data.data then
        for _,s in ipairs(data.data) do
            if s.playing<s.maxPlayers and s.id~=game.JobId then
                local p=s.playing
                local m=(sizeType=='Small' and p>=3 and p<=5)
                    or (sizeType=='Medium' and p>=13 and p<=17)
                    or (sizeType=='Large' and p>=20 and p<=23)
                    or (sizeType=='Any' and p>=1)
                if m then TeleportService:TeleportToPlaceInstance(game.PlaceId,s.id,LocalPlayer) return end
            end
        end
    end
    warn('[AzxRae] No server found: '..sizeType)
end

-- ─── DRAG HELPER ───────────────────────────────────────────────
local function makeDraggable(handle, frame)
    local dragging, dragStart, startPos = false, nil, nil
    handle.InputBegan:Connect(function(i)
        if i.UserInputType==Enum.UserInputType.MouseButton1 or i.UserInputType==Enum.UserInputType.Touch then
            dragging=true dragStart=i.Position
            startPos=frame.Position
        end
    end)
    handle.InputEnded:Connect(function(i)
        if i.UserInputType==Enum.UserInputType.MouseButton1 or i.UserInputType==Enum.UserInputType.Touch then
            dragging=false
        end
    end)
    UserInputService.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType==Enum.UserInputType.MouseMovement or i.UserInputType==Enum.UserInputType.Touch) then
            local delta=i.Position-dragStart
            frame.Position=UDim2.new(
                startPos.X.Scale, startPos.X.Offset+delta.X,
                startPos.Y.Scale, startPos.Y.Offset+delta.Y
            )
        end
    end)
end

-- Slider control
local function makeSlider(parent,label,cfgKey,minV,maxV,step,suffix)
    local card=makeCard(parent,70)
    step=step or 1 suffix=suffix or ''

    local lbl=Instance.new('TextLabel')
    lbl.Size=UDim2.new(0.6,0,0,18)
    lbl.Position=UDim2.new(0,14,0,10)
    lbl.BackgroundTransparency=1
    lbl.Text=label
    lbl.TextColor3=T.text
    lbl.Font=Enum.Font.GothamSemibold
    lbl.TextSize=12
    lbl.TextXAlignment=Enum.TextXAlignment.Left
    lbl.Parent=card

    local valLbl=Instance.new('TextLabel')
    valLbl.Size=UDim2.new(0.35,0,0,18)
    valLbl.Position=UDim2.new(0.62,0,0,10)
    valLbl.BackgroundTransparency=1
    valLbl.Text=tostring(Cfg[cfgKey])..suffix
    valLbl.TextColor3=T.accentLight
    valLbl.Font=Enum.Font.GothamBold
    valLbl.TextSize=12
    valLbl.TextXAlignment=Enum.TextXAlignment.Right
    valLbl.Parent=card

    -- Track
    local track=Instance.new('Frame')
    track.Size=UDim2.new(1,-28,0,6)
    track.Position=UDim2.new(0,14,0,40)
    track.BackgroundColor3=T.sliderBar
    track.BorderSizePixel=0
    track.Parent=card
    corner(3,track)

    -- Fill
    local fill=Instance.new('Frame')
    local pct=math.clamp((Cfg[cfgKey]-minV)/(maxV-minV),0,1)
    fill.Size=UDim2.new(pct,0,1,0)
    fill.BackgroundColor3=T.sliderFill
    fill.BorderSizePixel=0
    fill.Parent=track
    corner(3,fill)

    local grad=Instance.new('UIGradient')
    grad.Color=ColorSequence.new{ColorSequenceKeypoint.new(0,T.accent),ColorSequenceKeypoint.new(1,T.accentGlow)}
    grad.Rotation=0
    grad.Parent=fill

    -- Knob
    local kn=Instance.new('Frame')
    kn.Size=UDim2.new(0,16,0,16)
    kn.Position=UDim2.new(pct,0,0.5,-8)
    kn.BackgroundColor3=Color3.fromRGB(255,255,255)
    kn.BorderSizePixel=0
    kn.ZIndex=2
    kn.Parent=track
    corner(8,kn)

    local function setVal(v)
        v=math.clamp(math.round(v/step)*step,minV,maxV)
        Cfg[cfgKey]=v
        local p2=math.clamp((v-minV)/(maxV-minV),0,1)
        tw(fill,0.1,{Size=UDim2.new(p2,0,1,0)})
        tw(kn,0.1,{Position=UDim2.new(p2,0,0.5,-8)})
        valLbl.Text=string.format((step<1 and '%.1f' or '%d'),v)..suffix
    end

    local sliding=false
    local btn=Instance.new('TextButton')
    btn.Size=UDim2.new(1,0,0,30)
    btn.Position=UDim2.new(0,0,0,34)
    btn.BackgroundTransparency=1
    btn.Text='' btn.ZIndex=3
    btn.Parent=card

    btn.InputBegan:Connect(function(i)
        if i.UserInputType==Enum.UserInputType.MouseButton1 or i.UserInputType==Enum.UserInputType.Touch then
            sliding=true
            local rx=math.clamp((i.Position.X-track.AbsolutePosition.X)/track.AbsoluteSize.X,0,1)
            setVal(minV+(maxV-minV)*rx)
        end
    end)
    btn.InputEnded:Connect(function(i)
        if i.UserInputType==Enum.UserInputType.MouseButton1 or i.UserInputType==Enum.UserInputType.Touch then
            sliding=false
        end
    end)
    UserInputService.InputChanged:Connect(function(i)
        if sliding and (i.UserInputType==Enum.UserInputType.MouseMovement or i.UserInputType==Enum.UserInputType.Touch) then
            local rx=math.clamp((i.Position.X-track.AbsolutePosition.X)/track.AbsoluteSize.X,0,1)
            setVal(minV+(maxV-minV)*rx)
        end
    end)

    return card
end

local tabs      = {}
local pages     = {}
local activeTab = nil

local function activateTab(tabName)
    if activeTab == tabName then return end
    activeTab = tabName
    for name, info in pairs(tabs) do
        local isActive = (name==tabName)
        tw(info.btn,0.2,{BackgroundColor3=isActive and T.tabActive or T.tab})
        tw(info.lbl,0.2,{TextColor3=isActive and T.accentLight or T.textDim})
        if info.dot then info.dot.Visible=isActive end
    end
    for name,page in pairs(pages) do
        if name==tabName then
            page.Visible=true
            page.Position=UDim2.new(0.05,0,0,0)
            page.BackgroundTransparency=1
            tw(page,0.25,{Position=UDim2.new(0,0,0,0)},Enum.EasingStyle.Quart)
        else
            page.Visible=false
        end
    end
end

-- ══════════════════════════════════════════════════════
-- KEYBOARD SHORTCUTS (PC)
-- ══════════════════════════════════════════════════════
UserInputService.InputBegan:Connect(function(input,gp)
    if gp then return end
    local k=input.KeyCode
    if k==Enum.KeyCode.F4 then
        guiOpen=not guiOpen
        Window.Visible=guiOpen
        if guiOpen then
            tw(Window,0.4,{Position=UDim2.new(0,30,0.5,-270)},Enum.EasingStyle.Back)
        end
    end
    if k==Enum.KeyCode.J then Cfg.InfJump=not Cfg.InfJump end
    if k==Enum.KeyCode.M then Cfg.AntiAfk=not Cfg.AntiAfk end
end)

-- Inf jump listen
UserInputService.JumpRequest:Connect(function()
    local hum=getHum()
    if Cfg.InfJump and hum and hum:GetState()~=Enum.HumanoidStateType.Jumping then
        hum:ChangeState(Enum.HumanoidStateType.Jumping)
    end
end)
